CREATE TABLE party_detail (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(300) NULL,
    email VARCHAR(150) NULL,
    mobilno VARCHAR(12) NULL,
    address VARCHAR(300) NULL,
    shipping_add VARCHAR(300) NULL,
    gstin_uin VARCHAR(20) NULL,
    created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


CREATE TABLE tax_invoice (
    id INT AUTO_INCREMENT PRIMARY KEY,
    serial_no VARCHAR(20) NULL UNIQUE,
    invoice_date DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
    name VARCHAR(300) NULL,
    email VARCHAR(150) NULL,
    mobile_no VARCHAR(12) NULL,
    address VARCHAR(300) NULL,
    shipping_add VARCHAR(300) NULL,
    gstin_uin VARCHAR(20) NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;



CREATE TABLE tax_invoice_items (
    id INT AUTO_INCREMENT PRIMARY KEY,

    invoice_id INT NOT NULL,

    item_no INT UNSIGNED NOT NULL,

    item_name VARCHAR(200) NULL,
    hsn_code VARCHAR(20) NULL,

    quantity DECIMAL(10,2) NOT NULL DEFAULT 0.00,

    unit VARCHAR(20) NULL,

    rate DECIMAL(12,2) NOT NULL DEFAULT 0.00,

    cgst_rate DECIMAL(5,2) NOT NULL DEFAULT 0.00,

    sgst_rate DECIMAL(5,2) NOT NULL DEFAULT 0.00,

    discount DECIMAL(10,2) NOT NULL DEFAULT 0.00,

    total_amount DECIMAL(12,2) NOT NULL DEFAULT 0.00,

    CONSTRAINT fk_tax_invoice
        FOREIGN KEY (invoice_id)
        REFERENCES tax_invoice(id)
        ON DELETE CASCADE,

    CONSTRAINT unique_invoice_item
        UNIQUE (invoice_id, item_no)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
