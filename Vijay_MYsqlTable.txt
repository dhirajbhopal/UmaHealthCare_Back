
CREATE TABLE departments (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    department_code VARCHAR(20) NOT NULL UNIQUE,
    department_name VARCHAR(100) NOT NULL,
    description TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;


CREATE TABLE staff (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    staff_code VARCHAR(30) NOT NULL UNIQUE,
    full_name VARCHAR(150) NOT NULL,
    designation VARCHAR(100) NOT NULL,
    department_id BIGINT UNSIGNED,
    mobile VARCHAR(20),
    email VARCHAR(150),
    gender ENUM('Male','Female','Other'),
    date_of_birth DATE,
    joining_date DATE,
    address TEXT,
    qualification VARCHAR(255),
    employment_type ENUM(
        'Permanent','Contract','Part Time','Visiting'
    ) DEFAULT 'Permanent',
    salary DECIMAL(12,2) DEFAULT 0,
    staff_status ENUM(
        'Active','Inactive','Resigned','Terminated'
    ) DEFAULT 'Active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_staff_department
    FOREIGN KEY (department_id)
    REFERENCES departments(id)
    ON DELETE SET NULL
) ENGINE=InnoDB;


CREATE TABLE doctors (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    doctor_code VARCHAR(30) NOT NULL UNIQUE,
    staff_id BIGINT UNSIGNED NOT NULL UNIQUE,
    specialization VARCHAR(150),
    registration_number VARCHAR(100),
    consultation_fee DECIMAL(10,2) DEFAULT 0,
    followup_fee DECIMAL(10,2) DEFAULT 0,
    opd_start_time TIME,
    opd_end_time TIME,
    is_online_booking BOOLEAN DEFAULT TRUE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_doctor_staff
    FOREIGN KEY (staff_id)
    REFERENCES staff(id)
    ON DELETE RESTRICT
) ENGINE=InnoDB;
